"""Extend G68 public snapshot, never consume installer request data."""
import sys,json,base64,hashlib,re,urllib.request,urllib.parse
from pathlib import Path
sys.path.insert(0,'/data/path-codecs')
from public_codecs import decode_blte,parse_install
from encoding import parse_encoding
from index_parser import parse_index

ROOT=Path('/data/g69-reference')
OLD=Path('/data/public-path-reference')
catalog=json.loads((OLD/'reference.json').read_text())
configs=json.loads((OLD/'configs.json').read_text())
fixtures={r['label']:(OLD/(r['label']+'.bin')).read_bytes() for r in catalog['receipts']}
paths=catalog['paths']
range_refs={}
new_receipts=[]
details=[]
reserved=0

class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self,*a,**kw):raise ValueError('redirect_forbidden')
opener=urllib.request.build_opener(urllib.request.ProxyHandler({}),NoRedirect())

def table(blob):
    ls=[s for s in blob.decode('utf-8-sig').splitlines() if s and not s.startswith('#')]
    names=[s.split('!')[0] for s in ls[0].split('|')]
    out=[]
    for s in ls[1:]:
        values=s.split('|')
        if len(values)!=len(names):raise ValueError('PSV_shape')
        out.append(dict(zip(names,values)))
    return out

def address(prefix,key,suffix=''):
    if not re.fullmatch(r'[a-f0-9]{32}',key) or not re.fullmatch(r'tpr/[a-zA-Z0-9_-]+(?:/[a-zA-Z0-9_-]+)*',prefix):
        raise ValueError('public_reference_shape')
    return '/'+prefix+'/'+key[:2]+'/'+key[2:4]+'/'+key+suffix

def fetch(path,label,cap,expected=None):
    global reserved
    if len(new_receipts)>=16 or not 0<cap<=1048576 or reserved+cap>4*1048576:
        raise ValueError('metadata_budget')
    reserved+=cap
    req=urllib.request.Request('https://us.cdn.blizzard.com'+path,
        headers={'User-Agent':'public-download-path-audit/1','Accept':'*/*','Accept-Encoding':'identity'})
    with opener.open(req,timeout=30) as r:
        if r.status!=200 or r.headers.get('Content-Encoding','identity')!='identity':
            raise ValueError('HTTP_contract')
        blob=r.read(cap+1)
        if len(blob)>cap or expected is not None and len(blob)!=expected:raise ValueError('body_length')
    fixtures[label]=blob
    new_receipts.append({'label':label,'url':'https://us.cdn.blizzard.com'+path,
        'bytes':len(blob),'sha256':hashlib.sha256(blob).hexdigest(),'status':200})
    return blob

def add(path,row):
    if row not in paths.setdefault(path,[]):paths[path].append(row)

for product in ('agent','bna'):
    cdns=table(fixtures[product+'-cdns'])
    cdn=next(x for x in cdns if x['Name']=='us')
    versions=table(fixtures[product+'-versions'])
    build=configs[product+'-build']
    # All regional config identities already declared in the fixed public table,
    # without pretending we fetched/validated each regional config body.
    for version in versions:
        regional=next((x for x in cdns if x['Name']==version['Region']),None)
        if regional is None:continue
        for kind,field in [('build','BuildConfig'),('cdn','CDNConfig'),('product','ProductConfig')]:
            key=version.get(field,'')
            if not re.fullmatch('[a-f0-9]{32}',key):continue
            prefix=regional['ConfigPath'] if kind=='product' else regional['Path']+'/config'
            add(address(prefix,key),{'kind':'public_regional_'+kind+'_config_reference',
                'product':product,'source_label':product+'-versions'})
    for kind in ('encoding','install','download','size','patch-index'):
        keys=build.get(kind,'').split()
        sizes=build.get(kind+'-size','').split()
        if len(keys)!=2 or len(sizes)!=2 or not all(x.isdigit() for x in sizes):
            raise ValueError('metadata_reference_shape')
        dec,enc=map(int,sizes)
        if not 0<dec<=1048576 or not 0<enc<=1048576:
            details.append({'product':product,'kind':kind,'available':False,'reason':'size_cap'})
            continue
        path=address(cdn['Path']+'/data',keys[1])
        label=product+'-'+kind+'-manifest'
        blob=fixtures.get(label)
        if blob is None:blob=fetch(path,label,enc,enc)
        plain=decode_blte(blob,dec,keys[1])
        if hashlib.md5(plain).hexdigest()!=keys[0]:raise ValueError('metadata_CKey')
        n=len(blob);hs=int.from_bytes(blob[4:8],'big')
        allowed=[{'start':0,'end':n-1,'basis':'entire_public_metadata_object'},
                 {'start':0,'end':7,'basis':'BLTE_minimum_header'}]
        if hs:
            allowed.append({'start':0,'end':hs-1,'basis':'BLTE_complete_header_and_chunk_table'})
            allowed.append({'start':8,'end':hs-1,'basis':'BLTE_chunk_table_after_minimum_header'})
            at=hs
            for i in range(int.from_bytes(blob[9:12],'big')):
                size=int.from_bytes(blob[12+i*24:16+i*24],'big')
                allowed.append({'start':at,'end':at+size-1,'basis':'complete_BLTE_encoded_chunk'})
                at+=size
            if at!=n:raise ValueError('BLTE_chunk_ranges')
        else:
            allowed.append({'start':8,'end':n-1,'basis':'complete_BLTE_chunkless_payload'})
        range_refs[path]={'ranges':allowed,'public_size':n,'source_label':label}
        if kind=='encoding':
            en=parse_encoding(plain)
            root_keys=set(build.get('root','').split())
            for spec in en['ekey_specs']:
                add(address(cdn['Path']+'/data',spec['ekey']),{
                    'kind':'public_encoding_EKey_entry','product':product,
                    'source_label':label,'public_encoded_bytes':spec['encoded_bytes']})
            for mapping in en['ckey_mappings']:
                if mapping['ckey'] in root_keys:
                    for ek in mapping['ekeys']:
                        add(address(cdn['Path']+'/data',ek),{
                            'kind':'public_root_via_verified_encoding','product':product,'source_label':label})
        if kind=='install':parse_install(plain)
        # Size and patch-index payload semantics not claimed, only BLTE structure.
        details.append({'product':product,'kind':kind,'available':True,'encoded_bytes':n,
            'BLTE_EKey_and_chunks_valid':True,'decoded_CKey_valid':True})
    # G68's five size-unknown ranges targeted agent archive paths. Fetch every
    # agent index from its public CDNConfig, not from a private observed key.
    if product=='agent':
        for i,key in enumerate(configs['agent-cdn']['archives'].split()):
            path=address(cdn['Path']+'/data',key)
            label='agent-archive-index-'+str(i)
            blob=fetch(path+'.index',label,131072)
            parsed=parse_index(blob,key,'archive_index')
            allowed=[{'start':r['offset'],'end':r['offset']+r['encoded_bytes']-1,
                      'basis':'complete_verified_archive_record'}
                     for r in parsed['entries']]
            range_refs[path]={'ranges':allowed,'public_size':None,'source_label':label}
            details.append({'product':'agent','kind':'archive_index','available':True,
                'records':len(parsed['entries']),'index_key_footer_TOC_and_blocks_valid':True})

catalog['receipts']+=new_receipts
catalog['range_references']=range_refs
catalog['extended_metadata_details']=details
catalog['archive_ranges_validated']=False # Only exact matches in observation can qualify.
catalog['schema']='independent-public-download-paths-g69-1'
(ROOT/'reference.json').write_text(json.dumps(catalog,separators=(',',':')))
(ROOT/'fixtures.json').write_text(json.dumps({k:base64.b64encode(v).decode()
                                          for k,v in fixtures.items()},separators=(',',':')))
(ROOT/'acquisition.json').write_text(json.dumps({'new_receipts':new_receipts,'details':details,
 'installer_input_used':False,'payload_program_archive_bodies_downloaded':False},indent=2))
print(json.dumps({'new_metadata_GETs':len(new_receipts),'new_body_bytes':sum(x['bytes'] for x in new_receipts),
                  'paths':len(paths),'objects_with_range_reference':len(range_refs),
                  'snapshot_json_bytes':(ROOT/'fixtures.json').stat().st_size,
                  'reference_json_bytes':(ROOT/'reference.json').stat().st_size}))
